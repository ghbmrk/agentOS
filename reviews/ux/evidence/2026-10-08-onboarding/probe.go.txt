type hooks struct {
	online, joinFails, trustFails bool
	box                           string
	sent                          []string
	joins                         []string
	trust                         []bool
	private                       []bool
}

func (h *hooks) Progress() localui.Progress {
	return localui.Progress{Phase: "ready", Online: h.online, Updated: true}
}
func (h *hooks) Networks() []string { return []string{"Neighbor", "Synthetic Home"} }
func (h *hooks) JoinNetwork(ssid, pw string) error {
	h.joins = append(h.joins, ssid)
	if h.joinFails {
		return errors.New("synthetic join failure")
	}
	h.online = true
	return nil
}
func (h *hooks) BoxNumber() string          { return h.box }
func (h *hooks) HostInfo() string           { return "Synthetic Host, 8 GB" }
func (h *hooks) Send(to, text string) error { h.sent = append(h.sent, text); return nil }
func (h *hooks) EnrollCode() (string, error) {
	return "otpauth://totp/AgentOS:AgentOS?secret=JBSWY3DPEHPK3PXP&issuer=AgentOS&digits=6&period=30", nil
}
func (h *hooks) ConfirmCode(c string) (bool, error) { return c == "123456", nil }
func (h *hooks) TrustHost(t bool) error {
	h.trust = append(h.trust, t)
	if h.trustFails {
		return errors.New("synthetic trust failure")
	}
	return nil
}
func (h *hooks) Providers() []localui.Provider {
	return []localui.Provider{{ID: "synthetic", Name: "Synthetic Provider", APIKey: true}}
}
func (h *hooks) ConnectAPIKey(string, string) error { return errors.New("synthetic rejection") }
func (h *hooks) StartDeviceCode(string) (string, string, error) {
	return "https://provider.example.test/", "SYNTHETIC", nil
}
func (h *hooks) SetPrivateOK(id string, b bool) error { h.private = append(h.private, b); return nil }
func (h *hooks) AlreadySetUp() bool                   { return false }
func (h *hooks) Finish(string) error                  { return nil }

type rig struct {
	s      *localui.Server
	h      *hooks
	cookie string
	store  *localui.MemStore
	now    time.Time
}

func newRig() *rig {
	r := &rig{h: &hooks{box: "+15550000100"}, cookie: strings.Repeat("a", 64), store: &localui.MemStore{}, now: time.Date(2026, 10, 8, 12, 0, 0, 0, time.UTC)}
	r.restart()
	return r
}
func (r *rig) restart() {
	var err error
	r.s, err = localui.New(localui.Config{AP: localui.APConfig{Iface: "wlan0", Uplink: "eth0", Addr: netip.MustParsePrefix("10.42.0.1/24"), SSID: "Synthetic-UX", Password: "SYNTHETIC-TEST-WIFI"}, Hooks: r.h, Store: r.store, SetupSecret: "SYNTHETIC-RESET-SECRET", Defaults: "synthetic defaults", Now: func() time.Time { return r.now }})
	if err != nil {
		panic(err)
	}
}
func (r *rig) req(method, path string, v url.Values) *httptest.ResponseRecorder {
	req := httptest.NewRequest(method, "http://10.42.0.1"+path, strings.NewReader(v.Encode()))
	req.RemoteAddr = "10.42.0.23:51000"
	req.AddCookie(&http.Cookie{Name: "agentos_setup", Value: r.cookie})
	req.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	w := httptest.NewRecorder()
	r.s.ServeHTTP(w, req)
	return w
}
func (r *rig) post(path string, v url.Values) {
	w := r.req("POST", path, v)
	if w.Code != 303 {
		panic(fmt.Sprintf("post %s: %d", path, w.Code))
	}
}
func save(r *rig, dir, name string) string {
	w := r.req("GET", "/setup", nil)
	if w.Code != 200 {
		panic(fmt.Sprintf("get %s: %d", name, w.Code))
	}
	os.WriteFile(filepath.Join(dir, name+".html"), w.Body.Bytes(), 0644)
	return w.Body.String()
}
